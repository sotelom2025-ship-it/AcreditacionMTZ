Cómo conectar Google Sheets
  1. Abrí tu Google Sheet y ve a Extensiones → Apps Script
  2. Pegá el siguiente código y guardalo:
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var sheet = ss.getSheetByName(data.hoja) || ss.insertSheet(data.hoja);
  if (data.accion === "agregar") {
    if (sheet.getLastRow() === 0) sheet.appendRow(data.encabezados);
    sheet.appendRow(data.fila);
    return ContentService.createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (data.accion === "leer") {
    var vals = sheet.getDataRange().getValues();
    return ContentService.createTextOutput(JSON.stringify({ok:true,data:vals}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
function doGet(e) { return doPost(e); }
